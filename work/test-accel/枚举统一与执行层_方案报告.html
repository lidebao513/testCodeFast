四、执行层核心设计（代码示意，落 enums.py）

class VerifyLayer(Enum):
    INTERFACE = "接口"   # 接口层：HTTP/requests 直接打接口
    UI = "UI"            # UI 层：Playwright 渲染页面交互

# 由来源类型推导默认执行层
def derive_verify_layer(ftype: str) -> VerifyLayer:
    # api(接口路由) / business(后端业务函数) → 接口层
    # page(页面) / ui / component(前端交互组件) → UI 层
    return VerifyLayer.INTERFACE if ftype in ("api", "business") else VerifyLayer.UI

# 执行模式选择 + 回退
def select_execution_mode(preferred: VerifyLayer, ftype: str, ui_available: bool) -> VerifyLayer:
    if preferred == VerifyLayer.INTERFACE:
        return VerifyLayer.INTERFACE          # 选接口层：直接走，无回退
    # preferred == UI：以下情况回退接口层验证
    if ftype in ("api", "business"):          # 本质无 UI 面 → 回退
        return VerifyLayer.INTERFACE
    if not ui_available:                       # 浏览器/Playwright 不可用 → 回退
        return VerifyLayer.INTERFACE
    return VerifyLayer.UI                      # 正常走 UI 层
回退触发条件有两类：① 测试点本质是接口型（api/business）；② UI 执行环境不可用（无浏览器）。 两者任一成立即转接口层验证，并可在报告中标注"layer_fallback: true"便于你追溯。
